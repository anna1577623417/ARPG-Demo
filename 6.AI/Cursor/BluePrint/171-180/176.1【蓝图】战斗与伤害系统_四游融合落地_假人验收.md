### §2.2 `PostureSystem`（只狼）

复用 `ResourcePool` —— 把 `ResourceType.Posture` 当成第二条 HP：

```csharp
// 2_Framework/Combat/Posture/PostureSystem.cs（新增）
public sealed class PostureSystem
{
    readonly IResourcePool _pool;
    readonly PostureProfileSO _profile;
    float _lastPoiseDamageTime;
    bool  _broken;

    public bool IsBroken => _broken;
    public float Current => _pool.GetCurrent(ResourceType.Posture);
    public float Max     => _pool.GetMax(ResourceType.Posture);

    public void OnPoiseDamage(float amount, IEntity target)
    {
        _pool.Drain(ResourceType.Posture, amount, out var newVal);
        _lastPoiseDamageTime = Time.time;

        if (!_broken && newVal <= 0f)
        {
            _broken = true;
            target.Tags.State.Add(GameplayStateTags.Executable);   // LOL 风格能力声明
            GlobalEventBus.Publish(new PostureBrokenEvent { Target = target });
        }
    }

    public void Tick(float dt)
    {
        // 韧性条被打满后 N 秒内不回，破势状态保持到 X 秒
        if (Time.time - _lastPoiseDamageTime < _profile.RegenDelay) return;
        _pool.Refill(ResourceType.Posture, _profile.RegenPerSec * dt, out var v);
        if (_broken && v >= Max * _profile.BreakRecoverRatio)
        {
            _broken = false;
            // 移除 Executable Tag 略
        }
    }
}
```

// 2_Framework/Combat/Aura/AuraEmitter.cs
public sealed class AuraEmitter : MonoBehaviour
{
    AuraDefinitionSO _def;
    IEntity _source;
    float _nextTick;
    float _expireTime;
    readonly Collider[] _buf = new Collider[16];

    public void Init(IEntity source, AuraDefinitionSO def) { _source = source; _def = def;
        _expireTime = def.Duration < 0 ? float.PositiveInfinity : Time.time + def.Duration; }

    void Update()
    {
        if (Time.time >= _expireTime) { Destroy(gameObject); return; }
        if (Time.time < _nextTick) return;
        _nextTick = Time.time + _def.TickInterval;

        var count = _def.Shape.Overlap(transform.position, transform.rotation, _buf);
        for (var i = 0; i < count; i++)
        {
            var recv = _buf[i].GetComponentInParent<IEffectReceiver>();
            if (recv == null || !FactionMatch(recv)) continue;
            EffectSystem.ApplyEffect(gameObject, recv, _def.PerTickEffect);
        }
    }
}

## §3 Runtime 层增量

### §3.1 `HitExecutor`（统一命中分发） — **新核心**

资料 §1851 提示 Hitbox 真正归宿是 `AttackInstance`，不是角色 / 武器 / 技能本身。所以新建 `HitExecutor` 作为"攻击实例"运行时：

```csharp
// 2_Framework/Combat/Hit/HitExecutor.cs
public sealed class HitExecutor
{
    readonly HashSet<IEntity> _alreadyHitThisInstance = new();   // 单实例去重

    public void Execute(
        IEntity source,
        HitShapeSO shape, Vector3 origin, Quaternion rot,
        DamageProfileSO damage,
        IReadOnlyList<ProcSO> procs = null,
        FactionFilter filter = FactionFilter.Hostile)
    {
        Span<Collider> buf = stackalloc Collider[16];
        var count = shape.OverlapNonAlloc(origin, rot, buf);

        for (var i = 0; i < count; i++)
        {
            var entity = buf[i].GetComponentInParent<IEntity>();
            if (entity == null || _alreadyHitThisInstance.Contains(entity)) continue;
            if (!FactionMatchers.Match(source, entity, filter)) continue;
            _alreadyHitThisInstance.Add(entity);

            // ① 弹反窗口拦截（只狼）
            if (entity.Tags.State.HasAll(CapabilityTags.Deflecting.Value))
            { DeflectResolver.Resolve(source, entity, damage); continue; }

            // ② 无敌窗口拦截
            if (entity.Tags.State.HasAll(GameplayStateTags.Invincible)) continue;

            // ③ 走 Pipeline
            var ctx = CombatContext.Build(source, entity);
            var hit = BuildHitContext(damage);
            var result = DamagePipeline.Compute(in ctx, in hit);

            // ④ 韧性条 / 破势（只狼）
            if (entity is IPostureBearer p)
                p.Posture.OnPoiseDamage(hit.PoiseDamage * GetResistance(entity, hit.DamageType), entity);

            // ⑤ 状态积累（艾尔登法环）
            StatusBuildUpResolver.Apply(entity, in hit);

            // ⑥ 落地伤害
            if (entity is IDamageable d) d.ReceiveDamage(in result, in ctx);

            // ⑦ Proc 连锁（雨中冒险 2）
            if (procs != null) ProcResolver.RollAndDispatch(source, entity, procs, in hit);
        }
    }
}
```

### §3.2 `ProcResolver`（雨 2 Proc 链 + Proc Coefficient 防 spam）

```csharp
public static class ProcResolver
{
    [ThreadStatic] static int s_chainDepth;
    const int MaxChainDepth = 4;     // 雨 2 用 Coefficient 而不是 depth，但保险起见加个硬上限

    public static void RollAndDispatch(IEntity src, IEntity tgt, IReadOnlyList<ProcSO> procs, in HitContext hit)
    {
        if (s_chainDepth >= MaxChainDepth) return;
        s_chainDepth++;
        try
        {
            for (var i = 0; i < procs.Count; i++)
            {
                var p = procs[i];
                if (UnityEngine.Random.value > p.Chance * p.ProcCoefficient) continue;
                if (p.Payload != null && tgt is IEffectReceiver recv)
                    EffectSystem.ApplyEffect(src.Transform.gameObject, recv, p.Payload);
                if (p.SpawnAura != null)
                    AuraSpawner.Spawn(src, tgt.Transform.position, p.SpawnAura);
                if (p.SpawnProjectile != null)
                    ProjectileSpawner.Spawn(src, tgt.Transform.position, (tgt.Transform.position - src.Transform.position).normalized, p.SpawnProjectile);
            }
        }
        finally { s_chainDepth--; }
    }
}
```

## §8 关键架构约束（不可越界）

1. **HitBox 归宿 = AttackInstance**（资料 §1851）—— 不挂 Collider 到角色/武器/技能，全部走 `HitExecutor` + `HitShapeSO.OverlapNonAlloc`。
2. **不新增 Tag 轨** —— LOL 语义层全部走 `Tags.Ability` + `Tags.State`，新增的只是预定义 Mask 常量。
3. **Player / AI 走同一管线**（资料 §1388）—— `HitExecutor.Execute(source: IEntity, …)`，玩家 Action 与 Boss Action 调用同一函数。Boss AI 只是 `Intent → SkillRoute`，**不直接** `boss.PlayAnimation`。
4. **EffectSystem 是统一终点**（资料 §1271）—— Buff / DOT / 状态爆发 / Aura tick / Proc Payload，全部走 `EffectSystem.ApplyEffect`。不允许新增并行通道。
5. **DamagePipeline.Stage 链可注入** —— W0 的 `DamageTypeReductionStage` 通过 `AddDefaultStage` 插入，不改 `Compute` 内部。
6. **不在 W0-W8 阶段碰**：BuffSO 复杂条件触发 / GAS 全套 / 元素反应（火+水）/ AI 感知 / Boss Phase / 网络同步 / Encounter Director。这些属于资料 §P1-P2，放到 180+ 切片。
7. **资料里的 SceneView 实时预览（§五-九）是 W9 选做** —— 不阻塞 W0-W8 的数值闭环验收。

## §10 三句话设计哲学

> 1. **Hitbox 属于 AttackInstance** —— 不属于角色、武器或技能；这是资料 §1851 划下的分水岭。
> 2. **Capability 不是分类，是事实声明** —— `HasTag(Control.Airborne)` 替代 `if (state==Jumping)`，系统扩展性的天花板由此抬高。
> 3. **EffectSystem 是单一终点** —— Proc/Aura/StatusBuildUp/Buff 全部最终转为 EffectDataSO 走老通道；不开第二条并行路径。
